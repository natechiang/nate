function englifyWord(token) { const [leading, base, trailing] = splitWordPunctuation(token); if (!base) return token; const original = base; const lower = base.toLowerCase(); let result = lower; // Case 1: vowel words (way) if (lower.endsWith("way")) { result = lower.slice(0, -3); } // Case 2: consonant words (ay) else if (lower.endsWith("ay")) { let core = lower.slice(0, -2); let bestGuess = core; // Try moving different sizes of consonant clusters for (let i = 1; i <= Math.min(3, core.length); i++) { let endCluster = core.slice(-i); let start = core.slice(0, -i); let candidate = endCluster + start; // ✅ smarter check: // valid if starts with consonant + has vowel later if ( !isVowel(candidate[0], 0) && [...candidate].some((c, idx) => isVowel(c, idx)) ) { bestGuess = candidate; } } result = bestGuess; } // restore capitalization result = preserveCase(original, result); return leading + result + trailing; }
